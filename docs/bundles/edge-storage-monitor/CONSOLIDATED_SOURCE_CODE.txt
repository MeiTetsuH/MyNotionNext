import { sanitizeDetails } from './details';
import type { HeartbeatDetails } from './details';

export const MINUTE = 60_000;
export const HOUR = 60 * MINUTE;
export const DAY = 24 * HOUR;
export const CHECK_IDS = ['dns_internal', 'dns_public', 'glance', 'tunnel', 'linux_ssh'] as const;
export const KNOWN_BENIGN_ALERTS = new Set(['file_station_vfs_warning']);
export type CheckId = typeof CHECK_IDS[number];
export type Probe = { ok: boolean; latencyMs: number | null; reason: string };
export interface HealthSummary {
  generatedAt: string | null;
  overall: 'ok' | 'warning' | 'unknown';
  cloudSuccessAt: string | null;
  cloudEvent: 'success' | 'failed' | 'started' | 'unknown';
  cloudEventAt: string | null;
  integrityAt: string | null;
  integrityOk: boolean;
  homesSnapshotAt: string | null;
  dockerSnapshotAt: string | null;
}
export interface Heartbeat {
  schemaVersion: 1;
  source: 'nas';
  sampledAt: string;
  checks: Record<CheckId, Probe>;
  health: HealthSummary;
  details?: HeartbeatDetails | null;
}
export interface StoredHeartbeat { receivedAt: number; data: Heartbeat }
export interface Incident { active: boolean; failures: number; lastNotified: number; since: number }
export interface MonitorState {
  lastSampledAt: string | null;
  incidents: Record<string, Incident>;
  lastCheckedAt: number;
}
export interface Notice { id: string; kind: 'alert' | 'reminder' | 'recovery'; label: string }

const labels: Record<string, string> = {

export function emptyState(): MonitorState {
  return { lastSampledAt: null, incidents: {}, lastCheckedAt: 0 };
}
export function timestamp(value: unknown): number | null {
  if (typeof value !== 'string') return null;
  const n = Date.parse(value);
  return Number.isFinite(n) ? n : null;
}
function record(value: unknown): value is Record<string, unknown> {
  return value !== null && typeof value === 'object' && !Array.isArray(value);
}
function dateOrNull(value: unknown): value is string | null {
  return value === null || (typeof value === 'string' && value.length <= 40 && timestamp(value) !== null);
}
export function validateHeartbeat(raw: unknown, now: number): Heartbeat {
  if (!record(raw) || raw.schemaVersion !== 1 || raw.source !== 'nas' || !record(raw.checks) || !record(raw.health)) throw new Error('invalid_payload');
  const sampled = timestamp(raw.sampledAt);
  if (sampled === null || Math.abs(now - sampled) > 10 * MINUTE) throw new Error('invalid_sample_time');
  const checks = {} as Record<CheckId, Probe>;
  for (const id of CHECK_IDS) {
    const v = raw.checks[id];
    if (!record(v) || typeof v.ok !== 'boolean' || !(v.latencyMs === null || (typeof v.latencyMs === 'number' && Number.isFinite(v.latencyMs) && v.latencyMs >= 0 && v.latencyMs <= 120000))) throw new Error('invalid_probe');
    if (typeof v.reason !== 'string' || !/^[a-z_]{1,40}$/.test(v.reason)) throw new Error('invalid_reason');
    checks[id] = { ok: v.ok, latencyMs: v.latencyMs, reason: v.reason };
  }
  const h = raw.health;
  for (const k of ['generatedAt', 'cloudSuccessAt', 'cloudEventAt', 'integrityAt', 'homesSnapshotAt', 'dockerSnapshotAt']) {
    if (!dateOrNull(h[k]) || (typeof h[k] === 'string' && timestamp(h[k])! > now + 10 * MINUTE)) throw new Error('invalid_health_time');
  }
  if (!['ok', 'warning', 'unknown'].includes(String(h.overall)) || !['success', 'failed', 'started', 'unknown'].includes(String(h.cloudEvent)) || typeof h.integrityOk !== 'boolean') throw new Error('invalid_health');
  // Copy only the approved fields; arbitrary report content never reaches storage.
  return { schemaVersion: 1, source: 'nas', sampledAt: new Date(sampled).toISOString(), checks, health: {
    generatedAt: h.generatedAt as string | null, overall: h.overall as HealthSummary['overall'],
    cloudSuccessAt: h.cloudSuccessAt as string | null, cloudEvent: h.cloudEvent as HealthSummary['cloudEvent'],
    cloudEventAt: h.cloudEventAt as string | null, integrityAt: h.integrityAt as string | null,
    integrityOk: h.integrityOk, homesSnapshotAt: h.homesSnapshotAt as string | null,
    dockerSnapshotAt: h.dockerSnapshotAt as string | null,
  }, details: sanitizeDetails(raw.details, now) };
}
function tooOld(value: string | null, now: number, maxAge: number): boolean {
  const t = timestamp(value);
  return t === null || now - t > maxAge;
}
export function evaluate(
  stored: StoredHeartbeat, previous: MonitorState, now: number, siteOk: boolean | null,
): { state: MonitorState; notices: Notice[] } {
  const state = structuredClone(previous);
  const notices: Notice[] = [];
  const fresh = now - stored.receivedAt <= 25 * MINUTE;
  const newSample = stored.data.sampledAt !== state.lastSampledAt;
  const h = stored.data.health;
  const healthyReport = fresh && !tooOld(h.generatedAt, now, 36 * HOUR);
  const conditions: Record<string, { bad: boolean | null; threshold: number; sample: boolean }> = {
    heartbeat: { bad: !fresh, threshold: 1, sample: true },
    public_site: { bad: siteOk === null ? null : !siteOk, threshold: 3, sample: true },
    health: { bad: fresh ? !healthyReport : null, threshold: 1, sample: true },
  };
  for (const id of CHECK_IDS) conditions[id] = { bad: fresh ? !stored.data.checks[id].ok : null, threshold: 3, sample: newSample };
  const cloudBad = tooOld(h.cloudSuccessAt, now, 120 * HOUR) || h.cloudEvent === 'failed' || h.cloudEvent === 'unknown'
    || (h.cloudEvent === 'started' && tooOld(h.cloudEventAt, now, 24 * HOUR));
  const nasAlerts = stored.data.details?.nas?.alerts ?? [];
  const hasUnresolvedAlerts = nasAlerts.some(a => !KNOWN_BENIGN_ALERTS.has(a));
  const dailyBad = h.overall !== 'ok' && (hasUnresolvedAlerts || nasAlerts.length === 0);
  const backupChecks = {
    cloud_backup: cloudBad,
    backup_integrity: !h.integrityOk || tooOld(h.integrityAt, now, 40 * DAY),
    homes_snapshot: tooOld(h.homesSnapshotAt, now, 48 * HOUR),
    docker_snapshot: tooOld(h.dockerSnapshotAt, now, 48 * HOUR),
    daily_health: dailyBad,
  };
  for (const [id, bad] of Object.entries(backupChecks)) conditions[id] = { bad: healthyReport ? bad : null, threshold: 1, sample: true };
  for (const [id, c] of Object.entries(conditions)) {
    const incident = state.incidents[id] ?? { active: false, failures: 0, lastNotified: 0, since: 0 };
    state.incidents[id] = incident;
    // Missing/stale observations must not be reported as a recovery.
    if (c.bad === null) continue;
    if (!c.bad) {
      if (incident.active) notices.push({ id, kind: 'recovery', label: labels[id] });
      incident.active = false; incident.failures = 0; incident.since = 0;
      continue;
    }
    if (c.sample) incident.failures = Math.min(incident.failures + 1, 1000);
    if (incident.failures < c.threshold) continue;
    if (!incident.active) {
      incident.active = true; incident.since = now; incident.lastNotified = now;
      notices.push({ id, kind: 'alert', label: labels[id] });
    } else if (now - incident.lastNotified >= DAY) {
      incident.lastNotified = now;
      notices.push({ id, kind: 'reminder', label: labels[id] });
    }
  }
  if (fresh) state.lastSampledAt = stored.data.sampledAt;
  state.lastCheckedAt = now;
  return { state, notices };
}
export function notificationText(notices: Notice[], now: number, drill = false): string {

const JWKS_URL = 'https://auth.corp.internal/cdn-cgi/access/certs';
let cachedJwks: { fetchedAt: number; keys: JsonWebKey[]; imported: Map<string, CryptoKey> } | null = null;
let lastJwksFetch = 0;

function base64UrlToBytes(input: string): Uint8Array {
  const padded = input.replace(/-/g, '+').replace(/_/g, '/') + '==='.slice((input.length + 3) % 4);
  const bin = atob(padded);
  const out = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
  return out;
}

export async function verifyAccessJwt(request: Request, expectedAud?: string): Promise<string | null> {
  const token = request.headers.get('Cf-Access-Jwt-Assertion');
  if (!token) return null;
  const parts = token.split('.');
  if (parts.length !== 3) return null;
  try {
    const header = JSON.parse(new TextDecoder().decode(base64UrlToBytes(parts[0]))) as { kid?: string; alg?: string };
    const payload = JSON.parse(new TextDecoder().decode(base64UrlToBytes(parts[1]))) as {
      iss?: string;
      aud?: string | string[];
      exp?: number;
      nbf?: number;
      email?: string;
    };
    if (header.alg !== 'RS256' || !header.kid) return null;
    if (payload.iss !== 'https://auth.corp.internal') return null;
    const nowSec = Math.floor(Date.now() / 1000);
    if (typeof payload.exp !== 'number' || payload.exp <= nowSec - 30) return null;
    if (typeof payload.nbf === 'number' && payload.nbf > nowSec + 60) return null;
    if (expectedAud) {
      const audList = Array.isArray(payload.aud) ? payload.aud : typeof payload.aud === 'string' ? [payload.aud] : [];
      if (!audList.includes(expectedAud)) return null;
    }
    const nowMs = Date.now();
    if (!cachedJwks || nowMs - cachedJwks.fetchedAt > 10 * 60_000) {
      if (nowMs - lastJwksFetch >= 60_000) {
        lastJwksFetch = nowMs;
        const res = await fetch(JWKS_URL, { signal: AbortSignal.timeout(5000) });
        if (res.ok) {
          const body = (await res.json()) as { keys?: JsonWebKey[] };
          if (Array.isArray(body.keys)) {
            cachedJwks = { fetchedAt: nowMs, keys: body.keys, imported: new Map() };
          }
        }
      }
      if (!cachedJwks) return null;
    }
    let jwk = cachedJwks.keys.find(k => (k as { kid?: string }).kid === header.kid);
    if (!jwk && nowMs - lastJwksFetch >= 60_000) {
      // Force refresh if kid not found and cooled down (handles key rotation without amplification)
      lastJwksFetch = nowMs;
      const res = await fetch(JWKS_URL, { signal: AbortSignal.timeout(5000) });
      if (res.ok) {
        const body = (await res.json()) as { keys?: JsonWebKey[] };
        if (Array.isArray(body.keys)) {
          cachedJwks = { fetchedAt: nowMs, keys: body.keys, imported: new Map() };
          jwk = cachedJwks.keys.find(k => (k as { kid?: string }).kid === header.kid);
        }
      }
    }
    if (!jwk) return null;
    let key = cachedJwks.imported.get(header.kid);
    if (!key) {
      key = await crypto.subtle.importKey('jwk', jwk, { name: 'RSASSA-PKCS1-v1_5', hash: 'SHA-256' }, false, ['verify']);
      cachedJwks.imported.set(header.kid, key);
    }
    const data = new TextEncoder().encode(`${parts[0]}.${parts[1]}`);
    const sig = base64UrlToBytes(parts[2]);
    const valid = await crypto.subtle.verify('RSASSA-PKCS1-v1_5', key, sig, data);
    if (!valid) return null;
    return payload.email ?? request.headers.get('Cf-Access-Authenticated-User-Email') ?? 'access-user';
  } catch {
    return null;
  }
}

export async function probeExternalService(target: ExternalTarget): Promise<ExternalServiceProbe> {
  const started = Date.now();
  try {
    const res = await fetch(target.url, {
      method: 'GET',
      redirect: 'manual',
      signal: AbortSignal.timeout(5000),
      headers: {
        'User-Agent': 'Mozilla/5.0 (compatible; HomelabMonitor/1.0)',
        'Cache-Control': 'no-cache',
      },
    });
    const latencyMs = Date.now() - started;
    await res.body?.cancel();
    const status = res.status;
    const location = res.headers.get('location') ?? '';
    const vercelMitigated = res.headers.get('x-vercel-mitigated');
    const vercelId = res.headers.get('x-vercel-id');
    const cfRay = res.headers.get('cf-ray');
    const edge = vercelId ? vercelId.split('::')[0] : cfRay ? cfRay.split('-')[1] ?? 'CF' : null;

    if (status >= 200 && status < 300) {
      return { ...target, ok: true, status, mode: 'ok', modeLabel: '200 在线', latencyMs, edge };
    }
    if (status === 429 && vercelMitigated === 'challenge') {
      return { ...target, ok: true, status, mode: 'waf_challenge', modeLabel: 'Vercel WAF 防护中', latencyMs, edge };
    }
    if ((status === 301 || status === 302 || status === 307 || status === 308) && location.includes('cloudflareaccess.com')) {
      return { ...target, ok: true, status, mode: 'access_gate', modeLabel: 'CF Access 保护中', latencyMs, edge };
    }
    if (status === 401 || ((status === 301 || status === 302 || status === 307 || status === 308) && (location.includes('/login') || location.includes('/signin') || location.includes('sso-api')))) {
      return { ...target, ok: true, status, mode: 'auth_gate', modeLabel: '登录/SSO 门禁', latencyMs, edge };
    }
    if (status === 301 || status === 302 || status === 307 || status === 308) {
      return { ...target, ok: true, status, mode: 'redirect', modeLabel: `${status} 跳转正常`, latencyMs, edge };
    }
    return { ...target, ok: false, status, mode: 'error', modeLabel: `HTTP ${status}`, latencyMs, edge };
  } catch {
    return { ...target, ok: false, status: null, mode: 'error', modeLabel: '连接超时/异常', latencyMs: Date.now() - started, edge: null };
  }
}

export const HISTORY_RETENTION_MS = 7 * 24 * 60 * 60_000;

/** Compact per-heartbeat record: {check: [ok, latencyMs]}. */
export function compactChecks(checks: Record<string, { ok: boolean; latencyMs: number | null }>): HeartbeatSample['c'] {
  const out: HeartbeatSample['c'] = {};
  for (const [id, p] of Object.entries(checks)) out[id] = [p.ok ? 1 : 0, p.latencyMs];
  return out;
}

export function compactSites(results: ExternalServiceProbe[]): SiteSample['r'] {
  return results.map(r => [r.id, r.ok ? 1 : 0, r.status, r.latencyMs, r.mode, r.edge ?? null]);
}

const MODE_LABELS: Record<ExternalServiceProbe['mode'], (status: number | null) => string> = {
  ok: s => `${s ?? 200} 在线`,
  waf_challenge: () => 'Vercel WAF 防护中',
  access_gate: () => 'CF Access 保护中',
  auth_gate: () => '登录门禁正常',
  redirect: s => `${s} 跳转正常`,
  error: s => (s ? `HTTP ${s}` : '连接超时/异常'),
};

/** Rebuild probe cards from the most recent stored cron sample (no live fetch). */
export function servicesFromSample(sample: SiteSample | null): ExternalServiceProbe[] {
  const byId = new Map((sample?.r ?? []).map(r => [r[0], r]));
  return EXTERNAL_TARGETS.map(t => {
    const r = byId.get(t.id);
    if (!r) return { ...t, ok: false, status: null, mode: 'error' as const, modeLabel: '暂无探测数据', latencyMs: null, edge: null };
    const status = r[2];
    const ok = r[1] === 1;
    const mode = MODE_LABELS[r[4]] ? r[4] : (ok ? 'ok' : 'error');
    const modeLabel = MODE_LABELS[mode](status);
    const edge = typeof r[5] === 'string' && r[5].length <= 24 ? r[5] : null;
    return { ...t, ok, status, mode, modeLabel, latencyMs: r[3], edge };
  });
}

export interface DashboardInput {
  heartbeat: StoredHeartbeat | null;
  state: MonitorState;
  silencedUntil: number;
  authenticatedBy: DashboardPayload['authenticatedBy'];
  userEmail: string | null;
  liveProbe: boolean;
  history?: HeartbeatSample[];
  siteHistory?: SiteSample[];
  now?: number;
}

export async function buildDashboardPayload(input: DashboardInput): Promise<DashboardPayload> {
  const now = input.now ?? Date.now();
  const { heartbeat, state } = input;
  const siteHistory = input.siteHistory ?? [];
  let overall: DashboardPayload['evaluated']['overall'] = 'waiting';
  let freshHeartbeat = false;
  let heartbeatAgeSec: number | null = null;
  let activeIncidents: string[] = [];

  if (heartbeat) {
    heartbeatAgeSec = Math.max(0, Math.round((now - heartbeat.receivedAt) / 1000));
    freshHeartbeat = now - heartbeat.receivedAt <= 25 * 60_000;
    activeIncidents = Object.entries(state.incidents)
      .filter(([, inc]) => inc.active)
      .map(([id]) => id);
    const nasAlerts = heartbeat.data.details?.nas?.alerts ?? [];
    const hasUnresolvedAlerts = nasAlerts.some(a => !KNOWN_BENIGN_ALERTS.has(a));
    const healthWarning = heartbeat.data.health.overall !== 'ok' && (hasUnresolvedAlerts || nasAlerts.length === 0);
    if (activeIncidents.length > 0 || !freshHeartbeat) overall = 'alert';
    else if (healthWarning) overall = 'warning';
    else overall = 'ok';
  }

  const latestSites = siteHistory.length ? siteHistory[siteHistory.length - 1] : null;
  let externalServices: ExternalServiceProbe[];
  let sitesCheckedAt: number | null;
  if (input.liveProbe || !latestSites) {
    externalServices = await Promise.all(EXTERNAL_TARGETS.map(t => probeExternalService(t)));
    sitesCheckedAt = now;
  } else {
    externalServices = servicesFromSample(latestSites);
    sitesCheckedAt = latestSites.t;
  }

  return {
    serverTime: now,
    authenticatedBy: input.authenticatedBy,
    userEmail: input.userEmail,
    silencedUntil: input.silencedUntil,
    heartbeat,
    state,
    evaluated: { overall, freshHeartbeat, heartbeatAgeSec, activeIncidents },
    externalServices,
    sitesCheckedAt,
    history: input.history ?? [],
    siteHistory,
  };
}

function checkView(c){
  var h = hb(), e = DATA.evaluated, hs = h ? h.health : null, det = details();
  var v = { value:'—', unit:'', k:'idle', sub:'暂无数据' };
  var inc = c.incidents ? c.incidents.some(incident) : incident(c.id);
  if (!h) return v;
  if (c.id === 'heartbeat') {
    var minutes = e.heartbeatAgeSec == null ? null : e.heartbeatAgeSec / 60;
    v = { value: minutes == null ? '—' : String(Math.round(minutes)), unit:'分钟前', k: e.freshHeartbeat ? 'ok' : 'err', sub: '采样 ' + hm(h.sampledAt) };
  } else if (c.probe) {
    var p = h.checks[c.id];
    v = { value: p && p.latencyMs != null ? ms(p.latencyMs) : '—', unit:'ms', k: !e.freshHeartbeat ? 'idle' : p && p.ok ? 'ok' : 'warn', sub: p ? (p.ok ? '正常' : '失败：' + reasonText(p.reason)) : '无数据' };
  } else if (c.site) {
    var s = (DATA.externalServices || []).filter(function(x){ return x.id === c.site; })[0];
    v = { value: s && s.latencyMs != null ? ms(s.latencyMs) : '—', unit:'ms', k: s ? (s.ok ? 'ok' : 'warn') : 'idle', sub: s ? s.modeLabel : '无数据' };
  } else if (c.id === 'health') {
    var a = ageH(hs.generatedAt);
    v = { value: a == null ? '—' : String(Math.round(a)), unit:'小时前', k: a == null || a > 36 ? 'warn' : 'ok', sub: '采样 ' + mdhm(hs.generatedAt) };
  } else if (c.id === 'daily_health') {
    var alerts = det.nas && det.nas.alerts && det.nas.alerts.length ? det.nas.alerts : [];
    var codes = alerts.join(', ');
    var isBenign = alerts.length > 0 && alerts.every(function(a){ return a === 'file_station_vfs_warning'; });
    if (isBenign) {
      v = { value: '注意', unit: '', k: 'warn', sub: '告警项：' + codes + '（已知已降级）' };
    } else if (hs.overall === 'ok') {
      v = { value: '正常', unit: '', k: 'ok', sub: '全部子项正常' };
    } else {
      v = { value: hs.overall === 'warning' ? '注意' : '未知', unit: '', k: 'warn', sub: codes ? '告警项：' + codes : '请查看 NAS 详细摘要' };
    }
  } else if (c.id === 'cloud_backup') {
    var b = ageH(hs.cloudSuccessAt);
    var bad = b == null || b > 120 || hs.cloudEvent === 'failed' || hs.cloudEvent === 'unknown';
    v = { value: b == null ? '—' : String(Math.round(b)), unit:'小时前', k: bad ? 'warn' : 'ok', sub: '最近事件：' + ({success:'成功',failed:'失败',started:'运行中',unknown:'未知'})[hs.cloudEvent] };
  } else if (c.id === 'backup_integrity') {
    var g = ageH(hs.integrityAt);
    v = { value: hs.integrityOk ? '通过' : '失败', unit:'', k: hs.integrityOk && g != null && g < 40 * 24 ? 'ok' : 'warn', sub: '检查于 ' + mdhm(hs.integrityAt) };
  } else if (c.id === 'snapshots') {
    var ha = ageH(hs.homesSnapshotAt), da = ageH(hs.dockerSnapshotAt);
    var mx = ha == null || da == null ? null : Math.max(ha, da);
    var cnt = det.nas ? (det.nas.snapshots.homes.count + ' + ' + det.nas.snapshots.docker.count + ' 个') : '';
    v = { value: mx == null ? '—' : String(Math.round(mx)), unit:'小时前', k: mx == null || mx > 48 ? 'warn' : 'ok', sub: (cnt ? cnt + ' · ' : '') + '最新 ' + mdhm(hs.homesSnapshotAt) };
  }
  var alerts = det.nas && det.nas.alerts && det.nas.alerts.length ? det.nas.alerts : [];
  var isBenignDaily = (c.id === 'daily_health') && alerts.length > 0 && alerts.every(function(a){ return a === 'file_station_vfs_warning'; });
  if (inc && !isBenignDaily) v.k = 'err';
  return v;
}

function containerKind(c){ return c.status !== 'running' || c.oom ? 'err' : c.health && c.health !== 'healthy' ? 'warn' : 'ok'; }
function diskKind(d){
  if (d.status === 'sleeping_not_queried') return 'idle';
  if (d.status === 'ok') return d.tempC != null && d.tempC >= 55 ? 'warn' : 'ok';
  if (d.status === 'unknown') return 'warn';
  if (d.status === 'warning') return 'warn';
  return 'err';
}
function diskNote(d){
  if (d.status === 'sleeping_not_queried') return '休眠中，未唤醒读取';
  if (d.status === 'unknown') return '未读取到 SMART 数据';
  if (d.status === 'warning') return 'SMART 有告警项';
  if (d.status === 'ok') return '';
  return '状态：' + d.status;
}
function volKind(nas){ var ok = nas && typeof nas.volumeUsedPct === 'number' && isFinite(nas.volumeUsedPct); return !ok ? 'idle' : nas.volumeUsedPct >= 92 ? 'err' : nas.volumeUsedPct >= 85 ? 'warn' : 'ok'; }

function attentionItems(){
  var out = [], h = hb(), det = details();
  if (!h) { out.push({ k:'warn', t:'还没有收到 NAS 心跳', s:'等待 NAS 每 10 分钟上报', go:null }); return out; }
  var alerts = det.nas && det.nas.alerts && det.nas.alerts.length ? det.nas.alerts : [];
  var isBenignDaily = alerts.length > 0 && alerts.every(function(a){ return a === 'file_station_vfs_warning'; });
  CHECKS.forEach(function(c){
    var v = checkView(c);
    if (v.k === 'err' || v.k === 'warn') {
      var benign = (c.id === 'daily_health' && isBenignDaily);
      out.push({ k:v.k, t:c.name + (v.k === 'err' ? ' 告警中' : benign ? '（需留意）' : ' 异常'), s:v.sub, go:['check', c.id], benign: benign });
    }
  });
  (DATA.externalServices || []).forEach(function(s){ if (!s.ok && s.id !== 'glance_public') out.push({ k:'warn', t:s.name + ' 不可达', s:s.modeLabel, go:['site', s.id] }); });
  det.containers.forEach(function(c){ var k = containerKind(c); if (k !== 'ok') out.push({ k:k, t:'容器 ' + c.name + ' ' + c.status, s:'来自每日体检', go:['containers'] }); });
  if (det.nas) {
    det.nas.disks.forEach(function(d){ var k = diskKind(d); if (k === 'warn' || k === 'err') out.push({ k:k, t:'磁盘 ' + d.id + (d.tempC != null && d.tempC >= 55 && d.status === 'ok' ? ' 温度 ' + d.tempC + '°C' : ' ' + diskNote(d)), s:'来自每日体检', go:['disks'] }); });
    if (det.nas.volumeUsedPct != null && det.nas.volumeUsedPct >= 85) out.push({ k: det.nas.volumeUsedPct >= 92 ? 'err' : 'warn', t:'存储卷已用 ' + det.nas.volumeUsedPct + '%', s:'剩余 ' + (det.nas.volumeFreeGib != null ? nf(det.nas.volumeFreeGib) + ' GiB' : '—'), go:['storage'] });
    if (det.nas.memUsedPct != null && det.nas.memUsedPct >= 90) out.push({ k: det.nas.memUsedPct >= 95 ? 'err' : 'warn', t:'NAS 内存占用 ' + det.nas.memUsedPct + '%', s:'可用 ' + (det.nas.memAvailGib != null ? det.nas.memAvailGib + ' GiB' : '—'), go:['nas'] });
  }
  if (det.dns && det.dns.running === false) out.push({ k:'err', t:'AdGuard Home 未运行', s:'DNS 解析与过滤受影响', go:['dns'] });
  if (det.jellyfin && det.jellyfin.ok === false) out.push({ k:'warn', t:'Jellyfin 服务无响应', s:'端口 8808 本地 Ping 失败', go:['services'] });
  if (det.trendradarWeb && det.trendradarWeb.ok === false) out.push({ k:'warn', t:'TrendRadar 晚报 Web 离线', s:'端口 8811 响应失败', go:['services'] });
  if (det.tailscaleMesh && det.tailscaleMesh.running === false) out.push({ k:'warn', t:'Tailscale 组网未运行', s:'CLI 状态异常', go:['dns'] });
  if (det.syncthing && det.syncthing.errors > 0) out.push({ k:'warn', t:'Syncthing 有 ' + det.syncthing.errors + ' 个同步错误', s:'请查看同步目录', go:['syncthing'] });
  if (det.tunnel && det.tunnel.haConnections != null && det.tunnel.haConnections < 2) out.push({ k:'warn', t:'隧道仅 ' + det.tunnel.haConnections + ' 个连接', s:'建议保持多连接', go:['tunnel'] });
  if (det.news && det.news.degraded) out.push({ k:'warn', t:'晚报数据源降级', s:det.news.sourcesOk + '/' + det.news.sourcesTotal + ' 个源成功', go:['news'] });
  if (det.news && det.news.lastStatus === 'error') out.push({ k:'warn', t:'晚报导出任务报错', s:'请检查 NAS last-export.json', go:['news'] });
  if (DATA.silencedUntil > now()) out.push({ k:'warn', t:'告警已静默', s:'至 ' + mdhm(DATA.silencedUntil) + '，期间不发 Telegram', go:null });
  return out;
}

function jstAt(base, hh, mm, addDays){ var p = jstParts(base); return Date.UTC(p.y, p.mo, p.d + (addDays || 0), hh, mm) - 9 * H; }
function nextDaily(hh, mm, days){ var t = now(); for (var i = 0; i < 40; i++) { var c = jstAt(t, hh, mm, i); if (c > t && (!days || days.indexOf(jstParts(c).wd) >= 0)) return c; } return null; }
function nextMonthly(day, hh, mm){ var t = now(); for (var i = 0; i < 62; i++) { var c = jstAt(t, hh, mm, i); if (c > t && jstParts(c).d === day) return c; } return null; }

/** Random per-request nonce (base64). Workers expose Web Crypto globally. */
export function newNonce(): string {
  const bytes = crypto.getRandomValues(new Uint8Array(16));
  let s = '';
  for (const b of bytes) s += String.fromCharCode(b);
  return btoa(s);
}

const bool = (v: unknown): boolean | null => (typeof v === 'boolean' ? v : null);
function time(v: unknown, now: number): string | null {
  if (typeof v !== 'string' || v.length > 40) return null;
  const t = Date.parse(v);
  return Number.isFinite(t) && t <= now + 10 * 60_000 ? v : null;
}
function day(v: unknown): string | null {
  return typeof v === 'string' && /^\d{4}-\d{2}-\d{2}$/.test(v) ? v : null;
}
function snapshot(v: unknown, now: number): SnapshotDetail {
  const r = isRecord(v) ? v : {};
  return { count: num(r.count, 10_000), latestAt: time(r.latestAt, now) };
}

export async function authorized(request: Request, token: string): Promise<boolean> {
  if (!token || token.length < 32) return false;
  const value = request.headers.get('Authorization') ?? '';
  if (value.length > 256) return false;
  const enc = new TextEncoder();
  const [actual, expected] = await Promise.all([
    crypto.subtle.digest('SHA-256', enc.encode(value)),
    crypto.subtle.digest('SHA-256', enc.encode(`Bearer ${token}`)),
  ]);
  return timingSafeEqual(new Uint8Array(actual), new Uint8Array(expected));
}
export async function readLimited(response: Request | Response, limit: number): Promise<string> {
  if (!response.body) return '';
  const reader = response.body.getReader();
  const chunks: Uint8Array[] = [];
  let size = 0;
  try {
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      size += value.byteLength;
      if (size > limit) { await reader.cancel(); throw new Error('body_too_large'); }
      chunks.push(value);
    }
  } finally { reader.releaseLock(); }
  const bytes = new Uint8Array(size);
  let offset = 0;
  for (const chunk of chunks) { bytes.set(chunk, offset); offset += chunk.byteLength; }
  return new TextDecoder().decode(bytes);
}

async function send(env: Env, notices: Notice[], drill = false): Promise<number> {
  const response = await fetch(`https://api.telegram.org/bot${env.TELEGRAM_BOT_TOKEN}/sendMessage`, {
    method: 'POST', signal: AbortSignal.timeout(12000),

  });
  const raw: unknown = JSON.parse(await readLimited(response, 16384));

export async function runCheck(env: Env): Promise<Record<string, unknown>> {
  const now = Date.now();
  const owner = crypto.randomUUID();
  const acquired = await env.DB.prepare('INSERT INTO monitor_lease (id, owner, expires_at) VALUES (1, ?, ?) ON CONFLICT(id) DO UPDATE SET owner=excluded.owner, expires_at=excluded.expires_at WHERE monitor_lease.expires_at < ? RETURNING owner').bind(owner, now + 2 * MINUTE, now).first<{owner: string}>();
  if (!acquired) return { state: 'busy' };
  try {
    const control = await env.DB.prepare('SELECT silenced_until FROM monitor_control WHERE id = 1').first<{silenced_until: number}>();
    if ((control?.silenced_until ?? 0) > now) return { state: 'silenced', until: control!.silenced_until };
    const heartbeat = await readHeartbeat(env.DB);
    if (!heartbeat) return { state: 'waiting_for_first_heartbeat' };
    const siteOk = await siteProbe(env);
    const result = evaluate(heartbeat, await readState(env.DB), now, siteOk);
    // Advance notification state only after Telegram acknowledges delivery.
    const messageId = result.notices.length ? await send(env, result.notices) : null;
    await env.DB.prepare('INSERT INTO monitor_state (id, state_json, checked_at) VALUES (1, ?, ?) ON CONFLICT(id) DO UPDATE SET state_json=excluded.state_json, checked_at=excluded.checked_at').bind(JSON.stringify(result.state), now).run();
    console.log(JSON.stringify({ event: 'check_complete', noticeCount: result.notices.length, siteOk }));
    return { state: 'checked', noticeCount: result.notices.length, messageId, siteOk, checkedAt: now };
  } finally {
    await env.DB.prepare('DELETE FROM monitor_lease WHERE id = 1 AND owner = ?').bind(owner).run();
  }
}

def utc_now():
    return dt.datetime.now(dt.timezone.utc).isoformat(timespec='milliseconds')


def skip_name(data, offset):
    """Skip an encoded DNS name without following compression pointers."""
    for _ in range(128):
        if offset >= len(data):
            raise ValueError('short_dns_name')
        size = data[offset]
        if size & 0xC0 == 0xC0:
            if offset + 1 >= len(data):
                raise ValueError('short_dns_pointer')
            pointer = ((size & 0x3F) << 8) | data[offset + 1]
            if pointer >= len(data):
                raise ValueError('invalid_dns_pointer')
            return offset + 2
        if size & 0xC0:
            raise ValueError('invalid_dns_label')
        offset += 1
        if size == 0:
            return offset
        offset += size
    raise ValueError('long_dns_name')


def parse_dns(data, query_id):
    if len(data) < 12:
        raise ValueError('short_dns_header')
    ident, flags, questions, answers, authorities, _ = struct.unpack('!6H', data[:12])
    if ident != query_id or not flags & 0x8000 or questions != 1:
        raise ValueError('invalid_dns_response')
    if flags & 0x200:
        return {'truncated': True}
    pos = skip_name(data, 12) + 4
    if pos > len(data):
        raise ValueError('short_dns_question')
    ips = []
    has_soa = False
    for index in range(answers + authorities):
        pos = skip_name(data, pos)
        if pos + 10 > len(data):
            raise ValueError('short_dns_record')
        kind, klass, _, length = struct.unpack('!HHIH', data[pos:pos + 10])
        pos += 10
        if pos + length > len(data):
            raise ValueError('short_dns_rdata')
        if index < answers and kind == 1 and klass == 1 and length == 4:
            ips.append(socket.inet_ntoa(data[pos:pos + 4]))
        if index >= answers and kind == 6 and klass == 1 and length >= 22:
            has_soa = True
        pos += length
    return {'truncated': False, 'rcode': flags & 15, 'addresses': ips, 'hasSoa': has_soa}


def recv_exact(sock, size):
    chunks = []
    remaining = size
    while remaining:
        part = sock.recv(remaining)
        if not part:
            raise ValueError('short_dns_tcp')
        chunks.append(part)
        remaining -= len(part)
    return b''.join(chunks)


def dns_query(domain, expected_ip=None, expect_negative=False):
    query_id = secrets.randbelow(65536)
    labels = domain.encode('ascii').split(b'.')
    packet = struct.pack('!6H', query_id, 0x100, 1, 0, 0, 0)
    packet += b''.join(bytes([len(label)]) + label for label in labels) + b'\0' + struct.pack('!HH', 1, 1)
    with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as sock:
        sock.settimeout(TIMEOUT)
        sock.connect(('127.0.0.1', 53))
        sock.send(packet)
        result = parse_dns(sock.recv(65535), query_id)
    if result.get('truncated'):
        with socket.create_connection(('127.0.0.1', 53), timeout=TIMEOUT) as sock:
            sock.sendall(struct.pack('!H', len(packet)) + packet)
            length = struct.unpack('!H', recv_exact(sock, 2))[0]
            result = parse_dns(recv_exact(sock, length), query_id)
    if result.get('truncated'):
        return False, 'truncated_dns'
    if expect_negative:
        good = result['rcode'] == 3 or (result['rcode'] == 0 and not result['addresses'] and result['hasSoa'])
        return good, 'ok' if good else 'unexpected_dns_answer'
    good = result['rcode'] == 0 and expected_ip in result['addresses']
    return good, 'ok' if good else 'unexpected_dns_answer'

def http_probe(url, markers=()):
    req = urllib.request.Request(url, headers={'User-Agent': 'HomelabMonitor/1.0', 'Cache-Control': 'no-cache'})
    with _no_proxy_opener.open(req, timeout=TIMEOUT) as response:
        body = response.read(524289)
        if len(body) > 524288:
            return False, 'response_too_large'
        text = body.decode('utf-8', errors='replace')
        good = response.status == 200 and all(marker in text for marker in markers)
        return good, 'ok' if good else 'unexpected_http_response'

def timed_probe(function):
    started = time.monotonic()
    try:
        ok, reason = function()
    except (socket.timeout, TimeoutError):
        ok, reason = False, 'timeout'
    except urllib.error.HTTPError:
        ok, reason = False, 'http_error'
    except Exception:
        ok, reason = False, 'probe_error'
    return {'ok': ok, 'latencyMs': round((time.monotonic() - started) * 1000, 1), 'reason': reason}

def _num(value):
    return value if isinstance(value, (int, float)) and not isinstance(value, bool) else None


def _text(value):

def _ts(value):
    """Trim sub-millisecond digits (Docker reports nanoseconds) so JS Date.parse accepts it."""
    if not isinstance(value, str):
        return None
    value = re.sub(r'(\.\d{3})\d+', r'\1', value)
    return value if len(value) <= 40 else None


def _cpu_times():
    try:
        parts = Path('/proc/stat').read_text().splitlines()[0].split()
        if parts[0] != 'cpu' or len(parts) < 6:
            return None
        nums = [int(x) for x in parts[1:9]]
        return sum(nums), nums[3] + nums[4]
    except (OSError, ValueError, IndexError):
        return None


def live_host_metrics(cpu_before=None, cpu_after=None):
    """Read unprivileged real-time host telemetry from /proc and /volume1."""
    out = {'cpuPct': None, 'memUsedPct': None, 'memAvailGib': None, 'swapUsedPct': None,
           'netRxBytes': None, 'netTxBytes': None, 'volumeUsedPct': None, 'volumeFreeGib': None,
           'uptimeHours': None, 'load': None}
    if cpu_before and cpu_after and cpu_after[0] > cpu_before[0]:
        total_d = cpu_after[0] - cpu_before[0]
        idle_d = max(0, cpu_after[1] - cpu_before[1])
        out['cpuPct'] = round(max(0.0, min(100.0, (total_d - idle_d) / total_d * 100.0)), 1)
    try:
        mem = {}
        for line in Path('/proc/meminfo').read_text().splitlines():
            if ':' in line:
                k, v = line.split(':', 1)
                parts = v.split()
                if parts and parts[0].isdigit():
                    mem[k.strip()] = int(parts[0])
        if mem.get('MemTotal', 0) > 0 and 'MemAvailable' in mem:
            total_kb, avail_kb = mem['MemTotal'], mem['MemAvailable']
            out['memUsedPct'] = round((total_kb - avail_kb) / total_kb * 100.0, 1)
            out['memAvailGib'] = round(avail_kb / 1048576.0, 2)
        if mem.get('SwapTotal', 0) > 0 and 'SwapFree' in mem:
            st, sf = mem['SwapTotal'], mem['SwapFree']
            out['swapUsedPct'] = round((st - sf) / st * 100.0, 1)
    except OSError:
        pass
    try:
        rx_total, tx_total = 0, 0
        for line in Path('/proc/net/dev').read_text().splitlines()[2:]:
            if ':' not in line:
                continue
            iface, data = line.split(':', 1)
            iface = iface.strip()
            if iface in ('ovs_eth0', 'eth0', 'ovs_eth1', 'eth1'):
                fields = data.split()
                if len(fields) >= 9:
                    rx_total += int(fields[0])
                    tx_total += int(fields[8])
        if rx_total > 0 or tx_total > 0:
            out['netRxBytes'], out['netTxBytes'] = rx_total, tx_total
    except (OSError, ValueError):
        pass
    try:
        v = os.statvfs('/volume1')
        denom = v.f_blocks - v.f_bfree + v.f_bavail
        if denom > 0:
            out['volumeUsedPct'] = round(100.0 * (v.f_blocks - v.f_bfree) / denom, 1)
            out['volumeFreeGib'] = round(v.f_bavail * v.f_frsize / 2**30, 1)
    except OSError:
        pass
    try:
        out['uptimeHours'] = round(float(Path('/proc/uptime').read_text().split()[0]) / 3600.0, 1)
        out['load'] = [round(x, 2) for x in os.getloadavg()]
    except (OSError, ValueError, IndexError):
        pass
    return out

def syncthing_details():
    try:
        apikey = None
        device_names = {}
        xml_path = Path('/volume1/docker/syncthing/config.xml')
        if xml_path.is_file():
            try:
                root = ET.parse(str(xml_path)).getroot()
                apikey = (root.findtext('gui/apikey') or '').strip() or None
                for d in root.findall('device'):
                    did, dname = d.attrib.get('id'), d.attrib.get('name')
                    if did and dname:
                        device_names[did] = dname.strip()
            except ET.ParseError:
                text = xml_path.read_text(encoding='utf-8', errors='ignore')
                m = re.search(r'<apikey>([^<]+)</apikey>', text)
                if m:
                    apikey = m.group(1).strip()
        if not apikey:
            return None
        headers = {'X-API-Key': apikey, 'User-Agent': 'HomelabMonitor/1.0'}
        f_stat = {}
        try:
            req_f = urllib.request.Request('http://127.0.0.1:8810/rest/db/status?folder=fhtnc-uqrtx', headers=headers)
            with _no_proxy_opener.open(req_f, timeout=3) as r:
                f_stat = json.loads(r.read(8192))
        except Exception:
            pass
        completion_pct = None
        try:
            req_cp = urllib.request.Request('http://127.0.0.1:8810/rest/db/completion?folder=fhtnc-uqrtx', headers=headers)
            with _no_proxy_opener.open(req_cp, timeout=3) as r:
                cp_stat = json.loads(r.read(4096))
                if _num(cp_stat.get('completion')) is not None:
                    completion_pct = round(float(cp_stat['completion']), 1)
        except Exception:
            pass
        devices = []
        try:
            req_c = urllib.request.Request('http://127.0.0.1:8810/rest/system/connections', headers=headers)
            with _no_proxy_opener.open(req_c, timeout=3) as r:
                c_stat = json.loads(r.read(16384))
            for did, dev in c_stat.get('connections', {}).items():
                if isinstance(dev, dict):
                    addr = dev.get('address', '')
                    cfg_name = device_names.get(did)
                    name = cfg_name or ('Pixel (192.168.1.34)' if '192.168.1.34' in addr else dev.get('clientVersion') or 'device')
                    devices.append({
                        'name': name[:48],
                        'connected': dev.get('connected') is True,
                        'address': addr[:64],
                        'inBytes': _num(dev.get('inBytesTotal')),
                        'outBytes': _num(dev.get('outBytesTotal')),
                    })
        except Exception:
            pass
        return {
            'state': _text(f_stat.get('state')),
            'inSyncFiles': _num(f_stat.get('inSyncFiles')),
            'inSyncBytes': _num(f_stat.get('inSyncBytes')),

            'needBytes': _num(f_stat.get('needBytes')),
            'errors': _num(f_stat.get('errors')),
            'completionPct': completion_pct,
            'devices': devices[:6],
        }
    except Exception:
        return None


def cloudflared_details():
    try:
        req = urllib.request.Request('http://127.0.0.1:20241/metrics', headers={'User-Agent': 'HomelabMonitor/1.0'})
        with _no_proxy_opener.open(req, timeout=3) as r:
            text = r.read(65536).decode('utf-8', 'ignore')
        ha = 0
        locations = []
        rtt = None
        total = 0
        concurrent = 0
        for line in text.splitlines():
            if line.startswith('cloudflared_tunnel_ha_connections '):
                try: ha = int(float(line.split()[1]))
                except (ValueError, IndexError): pass
            elif 'cloudflared_tunnel_server_locations{' in line and line.strip().endswith(' 1'):
                m = re.search(r'edge_location="([^"]+)"', line)
                if m: locations.append(m.group(1)[:16])
            elif 'quic_client_smoothed_rtt{' in line:
                try:
                    val = float(line.split()[1])
                    rtt = val if rtt is None else min(rtt, val)
                except (ValueError, IndexError): pass
            elif line.startswith('cloudflared_tunnel_total_requests '):
                try: total = int(float(line.split()[1]))
                except (ValueError, IndexError): pass
            elif line.startswith('cloudflared_tunnel_concurrent_requests_per_tunnel '):
                try: concurrent = int(float(line.split()[1]))
                except (ValueError, IndexError): pass
        return {
            'haConnections': ha,
            'edgeLocations': locations[:8],
            'smoothedRttMs': round(rtt, 1) if rtt is not None else None,
            'totalRequests': total,
            'concurrentRequests': concurrent,
        }
    except Exception:
        return None

def tailscale_mesh_details():
    try:
        bin_path = '/var/packages/Tailscale/target/bin/tailscale'
        if not os.path.exists(bin_path):
            return None
        proc = subprocess.run([bin_path, 'status', '--json'], stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True, timeout=4)
        if proc.returncode != 0:
            return {'running': False, 'selfName': None, 'selfIp': None, 'peers': []}
        data = json.loads(proc.stdout)
        self_node = data.get('Self', {})
        self_ips = self_node.get('TailscaleIPs', [])
        self_ip = self_ips[0] if self_ips else None
        peers = []
        for _, p in data.get('Peer', {}).items():
            if isinstance(p, dict):
                p_ips = p.get('TailscaleIPs', [])
                peers.append({
                    'name': _text(p.get('HostName')) or 'peer',
                    'ip': p_ips[0] if p_ips else '',
                    'os': _text(p.get('OS')) or '',
                    'online': p.get('Online') is True,
                    'relay': _text(p.get('Relay')),
                })
        peers.sort(key=lambda x: (not x['online'], x['name']))
        return {
            'running': True,
            'selfName': _text(self_node.get('HostName')),
            'selfIp': self_ip,
            'peers': peers[:24],
        }
    except Exception:
        return None

def collect(config=None):
    # A unique reserved-domain query avoids treating an old DNS cache entry as a working upstream.
    cpu_before = _cpu_times()
    upstream_name = 'monitor-' + secrets.token_hex(8) + '.example.com'
    jobs = {
        'dns_internal': lambda: dns_query('nas.internal', expected_ip='192.168.1.13'),
        'dns_public': lambda: dns_query(upstream_name, expect_negative=True),
        'glance': lambda: http_probe('http://127.0.0.1:8802/', ('glance', '<title>Home</title>')),
        'tunnel': lambda: http_probe('http://127.0.0.1:20241/ready'),
        'linux_ssh': ssh_probe,
    }
    with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:
        futures = {name: pool.submit(timed_probe, job) for name, job in jobs.items()}
        ext_futures = {
            'dns': pool.submit(agh_details, config),
            'syncthing': pool.submit(syncthing_details),
            'tunnel': pool.submit(cloudflared_details),
            'jellyfin': pool.submit(jellyfin_details),
            'trendradar_web': pool.submit(trendradar_web_details),
            'tailscale': pool.submit(tailscale_mesh_details),
        }
        checks = {name: future.result() for name, future in futures.items()}
        precomputed = {k: f.result() for k, f in ext_futures.items()}
    cpu_after = _cpu_times()
    live = live_host_metrics(cpu_before, cpu_after)
    return {'schemaVersion': 1, 'source': 'nas', 'sampledAt': utc_now(), 'checks': checks, 'health': health_summary(),
            'details': details(config=config, live=live, precomputed=precomputed)}


def read_config(path):
    info = path.stat()
    if info.st_mode & (stat.S_IRWXG | stat.S_IRWXO):
        raise ValueError('config_permissions_too_open')
    config = json.loads(path.read_text())
    endpoint = config['endpoint']
    parsed = urllib.parse.urlparse(endpoint)
    if parsed.scheme != 'https' or parsed.hostname not in ('edge-monitor.corp.workers.dev', 'status.corp.internal') or parsed.path != '/heartbeat' or parsed.query or parsed.fragment or parsed.username or parsed.password:
        raise ValueError('invalid_endpoint')
    if not isinstance(config.get('token'), str) or len(config['token']) < 32:
        raise ValueError('invalid_token')
    return config


def post(payload, config):
    endpoints = [config['endpoint']]
    primary = config['endpoint']
    if 'edge-monitor.corp.workers.dev' in primary:
        endpoints.append(primary.replace('edge-monitor.corp.workers.dev', 'status.corp.internal'))
    elif 'status.corp.internal' in primary:
        endpoints.append(primary.replace('status.corp.internal', 'edge-monitor.corp.workers.dev'))

    data = json.dumps(payload).encode()
    headers = {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer ' + config['token'],
        'User-Agent': 'Mozilla/5.0 (compatible; HomelabMonitor/1.0)',
    }
    last_err = None
    for ep in endpoints:
        try:
            req = urllib.request.Request(ep, data=data, method='POST', headers=headers)
            with _no_proxy_opener.open(req, timeout=15) as resp:
                accepted = json.loads(resp.read(2048)).get('accepted') is True
                if resp.status == 202 and accepted:
                    return
        except Exception as e:
            last_err = e
            continue
    raise ValueError('heartbeat_not_accepted') from last_err


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--config', type=Path)
    parser.add_argument('--dry-run', action='store_true')
    args = parser.parse_args()
    try:
        config = None
        if args.config:
            config = read_config(args.config)
        elif not args.dry_run:
            parser.error('--config is required')
        payload = collect(config)
        if args.dry_run:
            print(json.dumps(payload, ensure_ascii=False, indent=2))
            return 0
        post(payload, config)
        return 0
    except Exception:
        # Never include request URLs, credentials, or raw report contents in task logs.
        print('Homelab heartbeat failed', file=sys.stderr)
        return 1


if __name__ == '__main__':
    sys.exit(main())

#!/usr/bin/env python3
"""Use authenticated monitor operations without placing secrets in argv."""
import argparse
import json
from pathlib import Path
import stat
import sys
import urllib.error
import urllib.request

BASE = 'https://edge-monitor.corp.workers.dev'
SECRETS = Path(__file__).resolve().parents[1] / 'private' / 'secrets.json'


class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('command', choices=['status', 'check', 'silence', 'test-alert', 'test-recovery'])
    parser.add_argument('--minutes', type=int, default=60)
    args = parser.parse_args()
    if SECRETS.stat().st_mode & (stat.S_IRWXG | stat.S_IRWXO):
        raise ValueError('Secret file must have owner-only permissions')
    token = json.loads(SECRETS.read_text())['ADMIN_TOKEN']
    path, payload = args.command, None
    if args.command == 'silence':
        if not 0 <= args.minutes <= 1440:
            parser.error('--minutes must be between 0 and 1440')
        payload = {'minutes': args.minutes}
    elif args.command.startswith('test-'):
        path, payload = 'test-notification', {'kind': args.command[5:]}
    request = urllib.request.Request(
        BASE + '/admin/' + path,
        method='GET' if args.command == 'status' else 'POST',
        data=json.dumps(payload).encode() if payload is not None else None,
        headers={
            'Authorization': 'Bearer ' + token,
            'Content-Type': 'application/json',
            'User-Agent': 'Mozilla/5.0 (compatible; HomelabMonitor/1.0)',
        },
    )
    opener = urllib.request.build_opener(urllib.request.ProxyHandler({}), NoRedirect())
    try:
        with opener.open(request, timeout=40) as response:
            result = json.loads(response.read(65536))
        print(json.dumps(result, ensure_ascii=False, indent=2))
        return 0
    except urllib.error.HTTPError as error:
        print('Monitor request failed with HTTP ' + str(error.code), file=sys.stderr)
        return 1
    except (OSError, ValueError):
        print('Monitor request failed', file=sys.stderr)
        return 1


if __name__ == '__main__':
    sys.exit(main())
